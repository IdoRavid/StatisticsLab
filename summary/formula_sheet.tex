\section*{Sequence Basics (Lab 1)}
%--------------------------------------------------------------------

\textbf{Chargaff's rule:} double-stranded DNA has A$\approx$T and C$\approx$G.\\
Consequence: proportion plots of A\&T track each other; so do C\&G.

\textbf{GC content} of a bin of size $b$:
$\text{gc}_k = (G_k + C_k)/b \in [0,1]$

%--------------------------------------------------------------------
\section*{Coverage \& the Poisson Model (Lec 2--3)}
%--------------------------------------------------------------------

$Y_i$ = number of reads starting at base $i$, for $i=1,\ldots,I$.\\
$J$ = total number of fragments in the experiment.

\textbf{Average coverage:}
\[
  \lambda = \frac{J}{I} = \frac{\text{total reads}}{\text{total bases}}, \qquad
  \text{avg distance} = \frac{1}{\lambda}
\]

\textbf{Uniform Poisson — 3 assumptions:}
\begin{enumerate}[leftmargin=*, itemsep=0pt, topsep=1pt]
  \item Identical distribution: every fragment has the same $p_i$
  \item Uniformity: $p_i = 1/I$ for all $i$
  \item Independence: fragments land independently
\end{enumerate}
(1)+(3): $Y_i \sim \text{Binomial}(J, p_i)$.
Adding (2) + Poisson approx ($J$ large, $1/I$ small):
\[
  Y_i \sim \text{Poisson}(\lambda), \quad
  \E[Y] = \Var[Y] = \lambda, \quad
  \text{SD}[Y] = \sqrt{\lambda}
\]
\textbf{Additive property:} $X_1,\ldots,X_K \sim \text{Pois}(\lambda)$ independent
$\Rightarrow X_1+\cdots+X_K \sim \text{Pois}(K\lambda)$. \\
$\Rightarrow$ Bin of size $b$: $Y_k \sim \text{Pois}(\lambda_k)$,\ $\lambda_k = \sum_{i \in k} p_i \cdot J$.

\textbf{VMR} (Variance-to-Mean Ratio):
\[
  \text{VMR} = \frac{\Var[Y]}{\E[Y]} = 1 \quad \text{under uniform Poisson}
\]

%--------------------------------------------------------------------
\section*{Overdispersion \& Variance Decomposition (Lec 3)}
%--------------------------------------------------------------------

Real data: VMR $\approx 14.7$ — uniform model fails. Why?

\textbf{Two-level model:} $Y_k \sim \text{Pois}(\lambda_k)$, each bin has its own rate.

\textbf{Law of Total Variance} (exact identity):
\[
  \Var(Y) = \underbrace{\E[\Var(Y\!\mid\!\lambda)]}_{\text{avg within-group}} + \underbrace{\Var(\E[Y\!\mid\!\lambda])}_{\text{variance of group means}}
\]
Applied here:
\[
  \boxed{\Var(Y_k) = \underbrace{\E[\lambda_k]}_{\bar\lambda\ (\text{Poisson noise})} + \underbrace{\Var(\lambda_k)}_{\text{rate heterogeneity}}}
\]
\[
  \text{VMR} = 1 + \frac{\Var(\lambda_k)}{\bar\lambda}
\]
VMR $=1$ iff all $\lambda_k$ equal. Spatial correlation of GC prevents $\Var(\lambda_k)$ from averaging out under binning — rates in nearby bins are similar.

\textbf{Robust estimators:} Center: median. Spread: IQR, MAD $= \med|x_i - \med(x)|$. Use instead of mean/SD when data are skewed or have outliers.

\textbf{B\&S (2012):} GC bias unimodal (peaks $\sim$50\%); driven by full-fragment GC (PCR cause); MAD 9.5 vs loess 10.8; fit per sample.

\textbf{From histogram} (midpoints $m_j$, counts $n_j$, total $n$):
\[
  \hat{\E}[Y] = \sum_j m_j \cdot \frac{n_j}{n}, \qquad
  \widehat{\Var}(Y) = \sum_j m_j^2 \cdot \frac{n_j}{n} - \hat{\E}[Y]^2
\]
Variance decomp plug-in: $\hat{\E}[\lambda_k] = \bar{Y}$,\ $\widehat{\Var}(\lambda_k) = \widehat{\Var}(Y) - \bar{Y}$.

\section*{Linear Regression for $f(\gc)$ (Lec 4)}
%--------------------------------------------------------------------

\textbf{Model:} $Y_k = \beta_0 + \beta_1\,\gc_k + \epsilon_k$, $\E[\epsilon_k]=0$

\textbf{OLS estimate:}
$\hat\beta_1 = \widehat{\text{Cov}}(\gc, Y)\,/\,\widehat{\Var}(\gc)$

\textbf{Residual plots}: residuals vs.\ $\gc$ and vs.\ fitted values should be flat. Fan shape $\Rightarrow$ heteroscedasticity. CIs from \texttt{lm} are \textbf{wrong} under misspecification.

$R^2 = 1 - \text{RSS}/\text{TSS}$, \quad $\text{RMSE} = \sqrt{\tfrac{1}{K}\sum_k r_k^2}$.

\textbf{Global polynomials}: sensitive to outliers — a few extreme points distort the entire fit.

%--------------------------------------------------------------------
\section*{Spline Regression (Lec 4)}
%--------------------------------------------------------------------

Key idea: divide the predictor into regions at \textbf{knots} $t_1,\ldots,t_m$; fit a polynomial in each piece; glue with smoothness constraints.

\textbf{Progression}: piecewise constant $\to$ piecewise linear (discontinuous) $\to$ continuous linear spline $\to$ cubic spline (continuous 1st \& 2nd derivatives).

\textbf{Cubic spline basis:}\ $\hat y = b_0 + b_1 x + b_2 x^2 + b_3 x^3 + \sum_j c_j(x-t_j)^3_+$, where $(u)_+=\max(u,0)$.

\textbf{B-spline} \texttt{bs(x,knots,degree=3)}: locally supported basis, same fit. More knots = more flexibility.

\textbf{Heteroscedasticity:} SD ratio $= \text{SD(resid, high-GC)}/\text{SD(resid, low-GC)} \gg 1$ $\Rightarrow$ variance grows with coverage.

\section*{Poisson GLM \& Quasi-Poisson (Lab 5)}
%--------------------------------------------------------------------

\textbf{Poisson GLM:} $\log\E[Y_k] = \beta_0 + \beta_1\,\gc_k$ (log link, counts).\\
Assumes $\Var(Y) = \E[Y]$ — still violated ($\phi \approx 14$).

\textbf{Quasi-Poisson:} same mean, $\Var(Y) = \phi\,\E[Y]$, $\phi$ estimated from data. More honest uncertainty.

\textbf{Comparing models:} $R^2$ for lm; McFadden pseudo-$R^2$ for GLM;
$\displaystyle\tilde R^2 = 1 - \frac{\log L(\hat{\text{model}})}{\log L(\text{null})}$.\\
Spearman $r$ is rank-based — robust to outliers, but misleading when there is no real signal (centromere paradox: Spearman $\approx 0.95$ despite $R^2 \approx 0$).

\textbf{Bin size tradeoff:} larger bins $\downarrow$ Poisson noise, $\uparrow$ GC signal blurring. Statistical elbow $\sim$5 kb. Biological constraint: genes 10--100 kb.

\section*{Copy Number Estimation (Labs 6--8, Yuval's notes)}
%--------------------------------------------------------------------

\textbf{Model:} $Y_k \sim \text{Pois}(\lambda_k)$,
\[
  \lambda_k = a_k \cdot f(\gc_k) \cdot \epsilon_k
\]
$a_k \in \{0, \tfrac{1}{2}, 1, \tfrac{3}{2}, 2,\ldots\}$ = copy number.
$f(\gc_k)$ = expected coverage at copy number 1.
$\epsilon_k$ = residual technical noise, $\E[\epsilon_k]=1$.

Identifiability: $\med_k\{a_k\}=1$,\ $\E_k[f(\gc_k)]=1$.

\textbf{Single-sample plug-in estimator} $\hat a_k^{(1)}$:
\[
  \hat a_k^{(1)} = \frac{Y_k}{\hat f(\gc_k)} \cdot \hat M, \quad \hat M = \frac{1}{\med_k\!\left\{\frac{Y_k}{\hat f(\gc_k)}\right\}}
\]
\textbf{Stabilized estimator} (reduces variance, increases bias for $a_k \neq 1$):
\[
  \hat a_k^{(1*)} = \frac{Y_k + c}{\hat f(\gc_k) + c}
\]

\textbf{Var decomp:} $\Var(\hat a_k) \approx \frac{1}{f(\gc_k)} + \frac{\sigma^2}{f(\gc_k)^2}$ (16\% Poisson, 84\% $\varepsilon$ noise).

\textbf{Two-sample model} (Yuval's note 2): decompose noise into shared $\gamma_k$ and sample-specific $\delta_k^T, \delta_k^N$:
\[
  Y_k^T = a_k \cdot f^T(\gc_k) \cdot \gamma_k \cdot \delta_k^T + \eta^T,\quad
  Y_k^N = f^N(\gc_k) \cdot \gamma_k \cdot \delta_k^N + \eta^N
\]
\textbf{Two-sample estimator:}
\[
  \hat a_k^{(2)} = \frac{Y_k^T/\hat f^T(\gc_k) + c}{Y_k^N/\hat f^N(\gc_k) + c}
  \;\approx\; a_k \cdot \frac{\delta_k^T}{\delta_k^N}
\]
Shared noise $\gamma_k$ cancels. Residual noise is ratio of sample-specific terms only — better than single-sample if $\gamma_k$ dominates.\\
\textbf{Fit separate GC spline per sample.} Ratio also cancels germline CNVs.

\textbf{Noise models:}
Additive: $\lambda_k = a_k f + \epsilon$ (offset independent of signal).
Multiplicative: $\lambda_k = a_k f \cdot \eta$, $\eta\sim N(1,\gamma^2)$ — error scales with signal; PCR amplifies proportionally $\Rightarrow$ more plausible.

\textbf{Poisson irreducible noise floor:}
\[
  \E\!\left[(Y_k - f(\gc_k))^2\right] = \E\!\left[(Y_k - \lambda_k)^2\right] \geq \E[\lambda_k] = \bar\lambda
\]
Training MSE $< \bar\lambda$ $\Rightarrow$ overfitting (memorizing noise).

\textbf{Key diagnostic}: corrected ratio vs position — flat at 1 under $H_0$.

\textbf{Aggregative model (Lab 7):} fit at 1 kb, aggregate predictions to 10 kb bins.

\section*{Hypothesis Testing \& Multiple Testing (Lab 9)}
%--------------------------------------------------------------------

\textbf{Statistic:} $S_k = \log_2(\hat a_k^{\text{2-sample}})$. Under $H_0$: $S_k \approx 0$.

\textbf{Negative controls}: locally stable bins (window median $\leq \log_2 1.10$, SD $\leq \log_2 1.15$), filtered on the \emph{ratio} so germline CNVs pass correctly.

\textbf{Empirical p-value:} $p_k = |\{j : |S_j| \geq |S_k|\}|/|\text{ctrl}|$. No distributional assumption.

\textbf{Calibration}: P-P plot of held-out CN=1 bin p-values should track the uniform diagonal.

\textbf{Bonferroni} (FWER): $\alpha'=\alpha/m$. Too conservative ($m\sim10^5$).\\
\textbf{Benjamini--Hochberg} (FDR): reject $H_{(k)}$ for $k \leq k^* = \max\{k: p_{(k)} \leq k\alpha/m\}$.\\
Controls $\E[\text{FP}/\text{sig}]\leq\alpha$. Preferred: tolerates a small known FP fraction for much greater power.
